```
import sympy as sp

# Symbols and original rational
a, b, c = sp.symbols('a b c')
num = (32*a**13*b**11*c**7
       + 64*a**10*b**6*c**10
       + 128*a**10*b**4*c**4
       + 32*a**9*b**14*c**9
       + 456*a**9*b**5*c**4
       + 64*a**6*b**9*c**12
       + 128*a**6*b**7*c**6
       + 456*a**5*b**8*c**6)
den = (4*a**5*b**7*c**3
       + 8*a**2*b**2*c**6
       - 57*a*c
       - 16*b**2
       - 57*b*c
       - 16*c**2
       - sp.Rational(43, 2))
expr = num/den

# Constraints: a+b+c=1, a*(1−a)=(c−2)^2/7
D = 7 + 2*c - 3*c**2
sqrtD = sp.sqrt(D)
a1 = (1 - c + sqrtD)/2
b1 = 1 - c - a1

# Substitute
subs_expr = sp.simplify(expr.subs({a: a1, b: b1}))

# Separate numerator and denominator in c
num_c = sp.simplify(sp.factor(sp.frac(subs_expr)[0]))
den_c = sp.simplify(sp.factor(sp.frac(subs_expr)[1]))

# Multiply through by den_c to clear denominator
poly_c = sp.simplify(sp.expand(num_c))

# Now poly_c is guaranteed a polynomial in c
poly = sp.Poly(poly_c, c)
print("Degree:", poly.degree())
print("Coeffs:", poly.all_coeffs())
print("Polynomial:", poly_c)

```

Key points:

* We extract numerator/denominator with `sp.frac`.
* Multiply through by the denominator (which symbolically vanishes on the constraint) so the result is a true polynomial.
* Finally, `sp.Poly` works without error because there’s no denominator left.
