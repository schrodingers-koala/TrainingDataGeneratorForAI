def solve_problem():
    import sympy as sp
    a, b, c, t = sp.symbols('a b c t')
    num = (32*a**13*b**11*c**7 + 64*a**10*b**6*c**10 + 128*a**10*b**4*c**4
           + 32*a**9*b**14*c**9 + 456*a**9*b**5*c**4 + 64*a**6*b**9*c**12
           + 128*a**6*b**7*c**6 + 456*a**5*b**8*c**6)
    den = (4*a**5*b**7*c**3 + 8*a**2*b**2*c**6 - 57*a*c
           - 16*b**2 - 57*b*c - 16*c**2 - sp.Rational(43, 2))
    eq1 = a + b + c - 1
    eq2 = a**2 + b**2 + c**2 - 4
    eq3 = num - t*den

    # eliminate b from eq1 and eq2
    R1 = sp.resultant(eq1, eq2, b)
    # eliminate a from R1 and eq3 → polynomial in c,t
    R2 = sp.resultant(R1, eq3, a)

    # factor and extract the linear-in-t factor
    for fac, mult in sp.factor_list(R2)[1]:
        P = sp.Poly(fac, t)
        if P.degree() == 1:
            # fac = coeff_t * t + coeff_0  ⇒  t = -coeff_0/coeff_t
            coeff_t = P.coeffs()[0]
            coeff_0 = P.coeffs()[1]
            return sp.simplify(-coeff_0/coeff_t)
    # if no linear factor found, raise
    raise ValueError("Failed to isolate t")
