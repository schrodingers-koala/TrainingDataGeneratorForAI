def solve_problem():
    import sympy as sp
    from sympy import Rational

    a, b, c = sp.symbols('a b c')
    # numerator and denominator
    num = (32*a**13*b**11*c**7
           + 64*a**10*b**6*c**10
           + 128*a**10*b**4*c**4
           + 32*a**9*b**14*c**9
           + 456*a**9*b**5*c**4
           + 64*a**6*b**9*c**12
           + 128*a**6*b**7*c**6
           + 456*a**5*b**8*c**6)
    den = (4*a**5*b**7*c**3
           + 8*a**2*b**2*c**6
           - 57*a*c
           - 16*b**2
           - 57*b*c
           - 16*c**2
           - Rational(43, 2))

    # express b in terms of a,c
    b_expr = 1 - a - c

    # solve a from the quadratic constraint at each sample point,
    # evaluate the ratio, then interpolate the resulting values to recover the polynomial.
    # pick 18 distinct rational samples for c in the valid range P = 7 + 2c - 3c^2 ≥ 0
    samples = [Rational(-5,2), Rational(-2), Rational(-3,2), Rational(-1), Rational(-1,2),
               Rational(0), Rational(1,2), Rational(1), Rational(3,2), Rational(5,2),
               Rational(7,2), Rational(8,2), Rational(9,2), Rational(10,2), Rational(11,2),
               Rational(12,2), Rational(13,2), Rational(14,2)]
    data = []
    for cs in samples:
        # quadratic in a: 2a^2 + (2c-2)a + (2c^2-2c-3)=0
        eq = 2*a**2 + (2*cs-2)*a + (2*cs**2 - 2*cs - 3)
        sols = sp.solve(eq, a)
        for sol in sols:
            bs = b_expr.subs({a: sol, c: cs})
            dval = den.subs({a: sol, b: bs, c: cs})
            if dval != 0:
                rval = sp.simplify(num.subs({a: sol, b: bs, c: cs})/dval)
                data.append((cs, rval))
                break
        if len(data) == 18:
            break

    # interpolate to recover the unique degree‐≤17 polynomial
    poly = sp.interpolate(data, c)
    return sp.simplify(poly)
