def solve_problem():
    import sympy as sp
    a, b, c, t = sp.symbols('a b c t')
    num = (32*a**13*b**11*c**7 + 64*a**10*b**6*c**10 + 128*a**10*b**4*c**4
           + 32*a**9*b**14*c**9 + 456*a**9*b**5*c**4 + 64*a**6*b**9*c**12
           + 128*a**6*b**7*c**6 + 456*a**5*b**8*c**6)
    den = (4*a**5*b**7*c**3 + 8*a**2*b**2*c**6 - 57*a*c - 16*b**2
           - 57*b*c - 16*c**2 - sp.Rational(43, 2))
    # Introduce t = num/den and eliminate a, b to express t as a polynomial in c
    G = sp.groebner(
        [a + b + c - 1, a**2 + b**2 + c**2 - 4, den*t - num],
        a, b, c, t,
        order='lex'
    )
    # Look for the basis element that is linear in t: t - H(c) = 0
    for g in G:
        poly_t = sp.Poly(g, t)
        if poly_t.degree() == 1:
            # isolate H(c)
            H = -g.subs(t, 0) / g.coeff(t, 1)
            return sp.simplify(H)
