def solve_problem():
    import sympy as sp
    a, b, c, t = sp.symbols('a b c t')
    
    num = (
        32*a**13*b**11*c**7
        + 64*a**10*b**6*c**10
        + 128*a**10*b**4*c**4
        + 32*a**9*b**14*c**9
        + 456*a**9*b**5*c**4
        + 64*a**6*b**9*c**12
        + 128*a**6*b**7*c**6
        + 456*a**5*b**8*c**6
    )
    den = (
        4*a**5*b**7*c**3
        + 8*a**2*b**2*c**6
        - 57*a*c
        - 16*b**2
        - 57*b*c
        - 16*c**2
        - sp.Rational(43, 2)
    )
    
    # clear rational coefficients by multiplying through by 2
    num2 = num
    den2 = den * 2
    
    # set up the three equations in the ideal
    eq1 = a + b + c - 1
    eq2 = a**2 + b**2 + c**2 - 4
    eq3 = num2 - t * den2
    
    # compute a lex Groebner basis over rational numbers
    G = sp.groebner([eq1, eq2, eq3], a, b, c, t, order='lex', domain=sp.QQ)
    
    # extract the elimination polynomial in c and t
    f_ct = None
    for poly in G.polynomials:
        if not any(v in poly.free_symbols for v in (a, b)):
            f_ct = poly
            break
    if f_ct is None:
        raise ValueError("No elimination polynomial found")
    
    # factor f_ct and find the factor linear in t
    for fac, _ in sp.factor_list(f_ct)[1]:
        P = sp.Poly(fac, t)
        if P.degree() == 1:
            coeff_t = P.coeff_monomial(t)
            coeff_0 = P.coeff_monomial(1)
            return sp.simplify(-coeff_0/coeff_t)
    
    raise ValueError("No linear factor in t found")
