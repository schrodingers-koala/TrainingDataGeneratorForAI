def solve_problem():
    import sympy as sp, random
    a, b, c, d = sp.symbols('a b c d')
    # define the original numerator and denominator
    num = 8*a**11*b**13*c**6 \
        + 5*a**10*b**11*c**9 \
        + 32*a**7*b**11*c**8 \
        + 664*a**7*b**6*c**4 \
        + 20*a**6*b**9*c**11 \
        + 464*a**6*b**7*c**4 \
        + 415*a**6*b**4*c**7 \
        + 290*a**5*b**5*c**7
    den = a**6*b**7*c**2 \
        + 4*a**2*b**5*c**4 \
        - 58*a*c \
        - 83*b**2 \
        - 58*b*c \
        - 83*c**2 \
        + 245

    # eliminate c using a + b + c = 1
    c_sub = 1 - a - b
    num_sub = num.subs(c, c_sub)
    den_sub = den.subs(c, c_sub)

    # the second constraint a^2 + b^2 + c^2 = 4
    f2 = a**2 + b**2 + c_sub**2 - 4

    # assume ratio = P(abc) is a polynomial in d = a*b*c of degree ≤ 5
    alpha = sp.symbols('alpha0:6')
    P = sum(alpha[i] * d**i for i in range(6))

    # collect 6 sample points on the curve
    data = []
    while len(data) < 6:
        ai = random.randint(1, 10)
        bi_sols = sp.solve(f2.subs(a, ai), b)
        for bi in bi_sols:
            try:
                bi = complex(bi)
                ci = 1 - ai - bi
                dv = den_sub.subs({a: ai, b: bi})
                if abs(dv) < 1e-6:
                    continue
                rv = complex(num_sub.subs({a: ai, b: bi}) / dv)
                di = ai * bi * ci
                data.append((di, rv))
                break
            except Exception:
                continue

    # build and solve the interpolation system
    eqs = [P.subs(d, di) - rv for di, rv in data]
    sol = sp.solve(eqs, alpha)

    # reconstruct the polynomial in a,b,c
    poly = P.subs(sol).subs(d, a*b*c)
    return sp.simplify(poly)
