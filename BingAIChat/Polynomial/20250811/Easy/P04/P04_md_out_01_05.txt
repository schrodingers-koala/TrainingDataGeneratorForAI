def solve_problem():
    import sympy as sp

    # declare symbols
    a, b, c, t = sp.symbols('a b c t')

    # original numerator and denominator
    num = (8*a**11*b**13*c**6
           + 5*a**10*b**11*c**9
           + 32*a**7*b**11*c**8
           + 664*a**7*b**6*c**4
           + 20*a**6*b**9*c**11
           + 464*a**6*b**7*c**4
           + 415*a**6*b**4*c**7
           + 290*a**5*b**5*c**7)
    den = (a**6*b**7*c**2
           + 4*a**2*b**5*c**4
           - 58*a*c
           - 83*b**2
           - 58*b*c
           - 83*c**2
           + 245)

    # substitute c from a+b+c=1
    c_sub = 1 - a - b

    # impose a^2 + b^2 + c^2 = 4
    f2_sub = a**2 + b**2 + c_sub**2 - 4

    # form the relation num - t*den with c substituted
    expr = num.subs(c, c_sub) - t*den.subs(c, c_sub)

    # eliminate b by taking resultant with f2_sub
    res_b = sp.resultant(expr, f2_sub, b)

    # res_b is now a polynomial in a and t; factor and extract coefficients of t
    P = sp.factor(res_b)
    coeffs = sp.Poly(P, t).all_coeffs()  # [coeff_t, coeff_const]

    # solve for t = -coeff_const/coeff_t
    t_sol = sp.simplify(-coeffs[1] / coeffs[0])
    return t_sol
