import sympy as sp

# symbols
a, b, c, t = sp.symbols('a b c t')

# constraints
f1 = a + b + c - 1
f2 = a**2 + b**2 + c**2 - 4

# build Groebner basis for constraints in a,b,c
G = sp.groebner([f1, f2], a, b, c, order='lex')

# numerator and denominator
num = (
    8*a**11*b**13*c**6
    + 5*a**10*b**11*c**9
    + 32*a**7*b**11*c**8
    + 664*a**7*b**6*c**4
    + 20*a**6*b**9*c**11
    + 464*a**6*b**7*c**4
    + 415*a**6*b**4*c**7
    + 290*a**5*b**5*c**7
)
den = (
    a**6*b**7*c**2
    + 4*a**2*b**5*c**4
    - 58*a*c
    - 83*b**2
    - 58*b*c
    - 83*c**2
    + 245
)

# form the difference num - t*den and reduce modulo the ideal <f1,f2>
expr = num - t*den
_, rem = G.reduce(expr)      # rem is a polynomial in a,b,c,t

# rem is linear in t: extract coefficients of t and constant term
poly_rem = sp.Poly(rem, t)
coeff_t = poly_rem.coeff_monomial(t)
coeff_const = poly_rem.coeff_monomial(1)

# solve for t: coeff_t * t + coeff_const = 0  =>  t = -coeff_const/coeff_t
solution_poly = sp.simplify(-coeff_const/coeff_t)

solve_flag = 1
poly = solution_poly
