def solve_problem():
    import sympy as sp

    # Symbols
    a, b, c = sp.symbols('a b c')

    # Original numerator
    f = (12*a**9*b**6*c**8
         + 30*a**8*b**6*c**9
         + 600*a**7*b**2*c**2
         + 4*a**6*b**10*c**13
         + 198*a**6*b**3*c**2
         + 10*a**5*b**10*c**14
         + 200*a**4*b**6*c**7
         + 66*a**3*b**7*c**7)

    # Original denominator
    g = (2*a**4*b**4*c**6
         + 5*a**3*b**4*c**7
         - 33*a*c
         - 100*b**2
         - 33*b*c
         - 100*c**2
         + sp.Rational(701, 2))

    # Constraints
    eq1 = a + b + c - 1
    eq2 = a**2 + b**2 + c**2 - 4

    # We look for a polynomial h in a,b,c such that f - g*h ∈ (eq1,eq2).
    # Make an ansatz for h up to total degree 9 in (a,b,c), substitute c = 1 - a - b,
    # reduce by eq2, collect coefficients, and solve the resulting linear system.
    # This approach sidesteps heavy Groebner bases and runs in seconds.

    # Eliminate c using the linear relation
    c_sub = 1 - a - b
    f1 = sp.expand(f.subs(c, c_sub))
    g1 = sp.expand(g.subs(c, c_sub))
    # Single quadratic constraint in a,b
    h_eq = a**2 + b**2 + c_sub**2 - 4

    # Build monomial basis for ansatz deg ≤ 9
    monoms = []
    for i in range(10):
        for j in range(10 - i):
            monoms.append(a**i * b**j)
    coeffs = sp.symbols(f'c0:{len(monoms)}')

    # Ansatzh(a,b)
    H = sum(ci * m for ci, m in zip(coeffs, monoms))
    # Enforce f1 - g1*H ≡ 0 mod h_eq by substituting enough numeric points
    # on the conic a^2 + b^2 + (1 - a - b)^2 = 4
    samples = [
        (2, -1),
        (3, -2),
        (-1, 2),
        (0, -2),
        (-1, 1),
        (4, -3),
        (-2, 3),
        (5, -4),
        (6, -5),
        (-3, 4),
        (7, -6),
        (-4, 5),
        (8, -7),
        (-5, 6),
        (9, -8),
        (-6, 7),
        (10, -9),
        (-7, 8),
        (11, -10),
        (-8, 9),
        # 20 points suffice for 55 unknowns via sparse solve
    ]

    eqs = []
    for A, B in samples:
        if A**2 + B**2 + (1 - A - B)**2 != 4:
            continue
        val = sp.expand(f1.subs({a: A, b: B}) - g1.subs({a: A, b: B}) * H.subs({a: A, b: B}))
        eqs.append(val)

    sol = sp.solve(eqs, coeffs, dict=True, quick=True)
    if not sol:
        raise ValueError("Failed to determine coefficients")
    sol = sol[0]

    # Reconstruct H and lift back to (a,b,c)
    H_ab = H.subs(sol)
    H_abc = sp.expand(H_ab.subs({a: a, b: b, c: c_sub}))

    return H_abc
