def solve_problem():
    import sympy as sp

    # symbols
    a, b, c = sp.symbols('a b c')
    t = sp.symbols('t')

    # original numerator and denominator
    f = (
        12*a**9*b**6*c**8 +
        30*a**8*b**6*c**9 +
        600*a**7*b**2*c**2 +
        4*a**6*b**10*c**13 +
        198*a**6*b**3*c**2 +
        10*a**5*b**10*c**14 +
        200*a**4*b**6*c**7 +
        66*a**3*b**7*c**7
    )
    g = (
        2*a**4*b**4*c**6 +
        5*a**3*b**4*c**7 -
        33*a*c -
        100*b**2 -
        33*b*c -
        100*c**2 +
        sp.Rational(701, 2)
    )

    # the constraints
    eq1 = a + b + c - 1
    eq2 = a**2 + b**2 + c**2 - 4

    # build a Groebner basis for just the two constraints
    G = sp.groebner([eq1, eq2], a, b, c, order='lex')

    # reduce f and g modulo that basis
    f_red = G.reduce(f)[1]
    g_red = G.reduce(g)[1]

    # under the ideal (eq1, eq2), f/g == f_red/g_red
    # so the simplified polynomial is just the quotient of those two reduced polys
    sol = sp.simplify(f_red / g_red)

    # expand to get a plain polynomial in a,b,c
    return sp.expand(sol)
