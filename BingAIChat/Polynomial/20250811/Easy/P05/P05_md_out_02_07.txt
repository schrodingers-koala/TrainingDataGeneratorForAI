def solve_problem():
    import sympy as sp
    from itertools import product

    # symbols
    a, b, c = sp.symbols('a b c')

    # original numerator and denominator
    f = (12*a**9*b**6*c**8
         + 30*a**8*b**6*c**9
         + 600*a**7*b**2*c**2
         + 4*a**6*b**10*c**13
         + 198*a**6*b**3*c**2
         + 10*a**5*b**10*c**14
         + 200*a**4*b**6*c**7
         + 66*a**3*b**7*c**7)
    g = (2*a**4*b**4*c**6
         + 5*a**3*b**4*c**7
         - 33*a*c
         - 100*b**2
         - 33*b*c
         - 100*c**2
         + sp.Rational(701, 2))

    # impose the two constraints
    eq1 = a + b + c - 1
    eq2 = a**2 + b**2 + c**2 - 4

    # build the rational function on the variety by eliminating nothing
    T = sp.simplify(f / g)

    # we expect the final answer to be a polynomial of total degree <= 3 in (a,b,c).
    # list all monomials of degree ≤ 3
    monoms = []
    for i in range(4):
        for j in range(4 - i):
            for k in range(4 - i - j):
                monoms.append(a**i * b**j * c**k)

    # collect enough sample points on the circle a+b+c=1, a^2+b^2+c^2=4
    samples = []
    for A, B, C in [
        (2, -1, 0),
        (-1, 2, 0),
        (0, 2, -1),
        (0, -1, 2),
        (2, 0, -1),
        (-1, 0, 2),
        (3, -2, 0),
        (-2, 3, 0),
        (0, 3, -2),
        (0, -2, 3),
        (3, 0, -2),
        (-2, 0, 3),
        (4, -3, 0),
        (-3, 4, 0),
        (0, 4, -3),
        (0, -3, 4),
        (4, 0, -3),
        (-3, 0, 4),
        (5, -4, 0),
        (-4, 5, 0),
        (0, 5, -4),
        (0, -4, 5),
        (5, 0, -4),
        (-4, 0, 5),
    ]:
        if A + B + C == 1 and A**2 + B**2 + C**2 == 4:
            samples.append((A, B, C))
        if len(samples) >= len(monoms):
            break

    # build the linear system M * coeffs = values
    M = []
    Y = []
    for A, B, C in samples:
        row = [m.subs({a: A, b: B, c: C}) for m in monoms]
        M.append(row)
        Y.append(T.subs({a: A, b: B, c: C}))

    M = sp.Matrix(M)
    Y = sp.Matrix(Y)

    # solve for the polynomial's coefficients
    coeffs = M.LUsolve(Y)

    # reconstruct the polynomial and return
    P = sum(c_i * m for c_i, m in zip(coeffs, monoms))
    return sp.expand(P)
