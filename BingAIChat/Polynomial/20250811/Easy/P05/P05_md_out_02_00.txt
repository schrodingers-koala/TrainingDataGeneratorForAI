def solve_problem():
    import sympy as sp

    a, b, c, t = sp.symbols('a b c t')

    # original numerator and denominator
    f = (12*a**9*b**6*c**8
         + 30*a**8*b**6*c**9
         + 600*a**7*b**2*c**2
         + 4*a**6*b**10*c**13
         + 198*a**6*b**3*c**2
         + 10*a**5*b**10*c**14
         + 200*a**4*b**6*c**7
         + 66*a**3*b**7*c**7)

    g = (2*a**4*b**4*c**6
         + 5*a**3*b**4*c**7
         - 33*a*c
         - 100*b**2
         - 33*b*c
         - 100*c**2
         + sp.Rational(701, 2))

    # the constraints
    eq1 = a + b + c - 1
    eq2 = a**2 + b**2 + c**2 - 4

    # eliminate a, b, c to get a single polynomial relation in t = f/g
    G = sp.groebner([eq1, eq2, f - t*g], a, b, c, t, order='lex')

    # pick off the univariate polynomial in t
    rel = next(p for p in G if p.free_symbols == {t})

    # solve for t and return the (unique) branch
    sol = sp.solve(rel, t)[0]
    return sp.simplify(sol)
