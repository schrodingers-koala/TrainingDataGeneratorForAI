def solve_problem():
    import sympy as sp

    a, b, c, t = sp.symbols('a b c t')
    c_sub = 1 - a - b

    # substitute c = 1 - a - b into numerator and denominator
    f = (
        12*a**9*b**6*c_sub**8 +
        30*a**8*b**6*c_sub**9 +
        600*a**7*b**2*c_sub**2 +
        4*a**6*b**10*c_sub**13 +
        198*a**6*b**3*c_sub**2 +
        10*a**5*b**10*c_sub**14 +
        200*a**4*b**6*c_sub**7 +
        66*a**3*b**7*c_sub**7
    )
    g = (
        2*a**4*b**4*c_sub**6 +
        5*a**3*b**4*c_sub**7 -
        33*a*c_sub -
        100*b**2 -
        33*b*c_sub -
        100*c_sub**2 +
        sp.Rational(701, 2)
    )

    # quadratic constraint in a,b
    h = a**2 + b**2 + c_sub**2 - 4

    # compute resultant eliminating b: resultant(f - t*g, h, b)
    R = sp.resultant(f - t*g, h, b)

    # factor the resultant and find the unique linear factor in t
    _, factors = sp.factor_list(R)
    for poly, exp in factors:
        if sp.degree(poly, t) == 1:
            sol = sp.solve(poly, t)[0]
            return sp.expand(sol)

    raise ValueError("No linear factor in t found")
