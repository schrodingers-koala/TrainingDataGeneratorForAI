def solve_problem():
    import sympy as sp
    a, b, c = sp.symbols('a b c')
    num = (
        32*a**12*b**9*c**9
        + 40*a**10*b**14*c**7
        + 32*a**10*b**10*c**13
        + 40*a**8*b**15*c**11
        + 148*a**6*b**2*c**7
        + 120*a**5*b**3*c**7
        + 185*a**4*b**7*c**5
        + 150*a**3*b**8*c**5
    )
    den = (
        8*a**8*b**7*c**2
        + 8*a**6*b**8*c**6
        - 30*a*c
        - 37*b**2
        - 30*b*c
        - 37*c**2
        + 103
    )
    G = sp.groebner([a + b + c - 1, a**2 + b**2 + c**2 - 4], a, b, c, order='lex')
    rem_num = G.reduce(num)[1]
    rem_den = G.reduce(den)[1]
    return sp.simplify(rem_num/rem_den)
