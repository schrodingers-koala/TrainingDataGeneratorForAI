def solve_problem():
    import sympy as sp
    a, b, c = sp.symbols('a b c')
    N = (2*a**12*b**12*c**9
         + 3*a**9*b**8*c**11
         + 2*a**8*b**12*c**8
         + 56*a**8*b**5*c**7
         + 70*a**7*b**6*c**7
         + 3*a**5*b**8*c**10
         + 56*a**4*b**5*c**6
         + 70*a**3*b**6*c**6)
    D = (2*a**6*b**7*c**2
         + 3*a**3*b**3*c**4
         - 70*a*c
         - 56*b**2
         - 70*b*c
         - 56*c**2
         + 119)
    data = []
    # sample rational c-values, solve for a,b, compute ratio, then interpolate
    for r in range(-8, 9):
        cv = sp.Rational(r, 1)
        sols = sp.solve([a + b + cv - 1,
                         a**2 + b**2 + cv**2 - 4],
                        [a, b], dict=True)
        vals = []
        for sol in sols:
            av = sol[a]; bv = sol[b]
            denom = D.subs({a: av, b: bv, c: cv})
            if denom == 0:
                continue
            vals.append(sp.simplify(N.subs({a: av, b: bv, c: cv})/denom))
        if not vals:
            continue
        # require both branches give same value
        if any(v != vals[0] for v in vals):
            raise ValueError("branch mismatch at c=%s" % cv)
        data.append((cv, vals[0]))
        if len(data) > 10:
            break
    poly = sp.interpolate(data, c)
    return sp.simplify(poly)
