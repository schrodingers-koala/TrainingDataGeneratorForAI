def solve_problem():
    import sympy as sp
    a, b, c = sp.symbols('a b c')
    # express a,b in terms of c using the two constraints
    S = 1 - c
    P = (2*c**2 - 2*c - 3)/2
    discr = S**2 - 4*P
    a1 = (S + sp.sqrt(discr)) / 2
    b1 = (S - sp.sqrt(discr)) / 2

    N = (
        2*a**12*b**12*c**9
        + 3*a**9*b**8*c**11
        + 2*a**8*b**12*c**8
        + 56*a**8*b**5*c**7
        + 70*a**7*b**6*c**7
        + 3*a**5*b**8*c**10
        + 56*a**4*b**5*c**6
        + 70*a**3*b**6*c**6
    )
    D = (
        2*a**6*b**7*c**2
        + 3*a**3*b**3*c**4
        - 70*a*c
        - 56*b**2
        - 70*b*c
        - 56*c**2
        + 119
    )

    # substitute the parameterization and simplify
    expr = sp.simplify(
        N.subs({a: a1, b: b1}) /
        D.subs({a: a1, b: b1})
    )
    return sp.simplify(expr)
