def solve_problem():
    import sympy as sp
    a, b, c, P = sp.symbols('a b c P')
    N = (2*a**12*b**12*c**9
         + 3*a**9*b**8*c**11
         + 2*a**8*b**12*c**8
         + 56*a**8*b**5*c**7
         + 70*a**7*b**6*c**7
         + 3*a**5*b**8*c**10
         + 56*a**4*b**5*c**6
         + 70*a**3*b**6*c**6)
    D = (2*a**6*b**7*c**2
         + 3*a**3*b**3*c**4
         - 70*a*c
         - 56*b**2
         - 70*b*c
         - 56*c**2
         + 119)
    G = sp.groebner(
        [a + b + c - 1,
         a**2 + b**2 + c**2 - 4,
         N - D*P],
        a, b, P, c,
        order='lex'
    ).basis
    # the last basis polynomial is linear in P: f(a,b,c,P)=0 ⇒ A(c)*P + B(c)=0
    f = G[-1]
    polyP = sp.Poly(f, P)
    solP = -polyP.coeff_monomial(P**0) / polyP.coeff_monomial(P**1)
    return sp.simplify(solP)
