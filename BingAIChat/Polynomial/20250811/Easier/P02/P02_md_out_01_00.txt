def solve_problem():
    import sympy as sp
    from sympy import simplify
    a, b, c = sp.symbols('a b c')
    
    # define numerator, denominator, and constraints
    num = 246*a**8*b*c**4 + 123*a**7*b**6*c + 468*a**7*b**2*c**4 \
        + 234*a**6*b**7*c - 6*a**6*b*c**4 - 3*a**5*b**6*c
    den = -78*a*c - 41*b**2 - 78*b*c - 41*c**2 + 46
    g1 = a + b + c - 1
    g2 = a**2 + b**2 + c**2 - 4

    # solve the two quadratic constraints for (a,b) in terms of c
    sol_ab = sp.solve([g1, g2], [a, b])
    # pick one branch; the final quotient will be the same on both
    a_expr, b_expr = sol_ab[0]

    # build the rational function restricted to the variety
    expr = simplify((num.subs({a: a_expr, b: b_expr}) /
                     den.subs({a: a_expr, b: b_expr})))

    # expr is now a univariate rational function in c; interpolate to get polynomial
    # assume it is actually a polynomial
    expr_poly = sp.simplify(sp.factor(expr.as_numer_denom()[0] /
                                      expr.as_numer_denom()[1]))
    return sp.simplify(expr_poly)

# When you run solve_problem(), it will return the unique polynomial in c representing
# the original rational function under the two constraints.
