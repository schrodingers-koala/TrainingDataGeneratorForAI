def solve_problem():
    import sympy as sp
    a, b, c = sp.symbols('a b c')
    num = 246*a**8*b*c**4 + 123*a**7*b**6*c + 468*a**7*b**2*c**4 + 234*a**6*b**7*c - 6*a**6*b*c**4 - 3*a**5*b**6*c
    den = -78*a*c - 41*b**2 - 78*b*c - 41*c**2 + 46
    g1 = a + b + c - 1
    g2 = a**2 + b**2 + c**2 - 4
    G = sp.groebner([g1, g2], a, b, c, order='lex')
    num_red = G.reduce(num)[1]
    den_red = G.reduce(den)[1]
    poly = sp.simplify(num_red/den_red)
    return poly
