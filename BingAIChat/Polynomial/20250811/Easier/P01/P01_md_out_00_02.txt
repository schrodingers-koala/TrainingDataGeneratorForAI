def solve_problem():
    import sympy as sp
    a, b, c = sp.symbols('a b c')
    expr = (304*a**9*b**5*c + 144*a**8*b**6*c + 456*a**7*b**8*c**8 + 216*a**6*b**9*c**8) / (-36*a*c - 76*b**2 - 36*b*c - 76*c**2 + 250)
    I = [a + b + c - 1, a**2 + b**2 + c**2 - 4]
    G = sp.groebner(I, a, b, c, order='lex')
    num, den = expr.as_numer_denom()
    num_red = G.reduce(num)[1]
    den_red = G.reduce(den)[1]
    poly = sp.simplify(num_red/den_red)
    return poly
