def solve_problem():
    import sympy as sp
    a, b, c, X = sp.symbols('a b c X')
    num = (
        450*a**9*b**3*c**5
        + 504*a**8*b**4*c**5
        - 48*a**7*b**3*c**5
        + 600*a**6*b**2*c**6
        + 672*a**5*b**3*c**6
        - 64*a**4*b**2*c**6
    )
    den = -84*a*c - 75*b**2 - 84*b*c - 75*c**2 + 166
    eqs = [
        den*X - num,
        a + b + c - 1,
        a**2 + b**2 + c**2 - 4
    ]
    # eliminate a, b, c to get a univariate polynomial in X
    elim = sp.eliminate(eqs, [a, b, c])[0]
    # solve for X
    roots = sp.solve(elim, X)
    # pick the (unique) solution expression
    return sp.simplify(roots[0])
