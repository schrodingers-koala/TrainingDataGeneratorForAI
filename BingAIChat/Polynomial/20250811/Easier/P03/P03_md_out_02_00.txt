def solve_problem():
    import sympy as sp

    # declare symbols
    a, b, c, X = sp.symbols('a b c X')

    # numerator and denominator of the original fraction
    num = (
        450*a**9*b**3*c**5
        + 504*a**8*b**4*c**5
        - 48*a**7*b**3*c**5
        + 600*a**6*b**2*c**6
        + 672*a**5*b**3*c**6
        - 64*a**4*b**2*c**6
    )
    den = -84*a*c - 75*b**2 - 84*b*c - 75*c**2 + 166

    # two constraint polynomials
    g1 = a + b + c - 1
    g2 = a**2 + b**2 + c**2 - 4

    # eliminate a by resultant of (den*X - num) and g1 wrt a
    r1 = sp.resultant(den*X - num, g1, a)

    # eliminate b by resultant of r1 and g2 wrt b
    r2 = sp.resultant(r1, g2, b)

    # factor to get the univariate polynomial in X
    poly_X = sp.factor(r2)

    # solve for X and pick the appropriate branch
    sols = sp.solve(poly_X, X)
    sol = sols[0]

    # return the simplified expression for X
    return sp.simplify(sol)
