def solve_problem():
    import sympy as sp
    a, b, c, X = sp.symbols('a b c X')
    num = 450*a**9*b**3*c**5 + 504*a**8*b**4*c**5 - 48*a**7*b**3*c**5 \
        + 600*a**6*b**2*c**6 + 672*a**5*b**3*c**6 - 64*a**4*b**2*c**6
    den = -84*a*c - 75*b**2 - 84*b*c - 75*c**2 + 166
    eqs = [
        den*X - num,
        a + b + c - 1,
        a**2 + b**2 + c**2 - 4
    ]
    # compute a lex Groebner basis eliminating a, b, c
    G = list(sp.groebner(eqs, X, a, b, c, order='lex'))
    # find the univariate polynomial in X
    poly_X = next(p for p in G if p.free_symbols == {X})
    # solve for X and return the unique solution
    sol = sp.solve(poly_X, X)
    return sp.simplify(sol[0])
