def solve_problem():
    import sympy as sp
    a, b, c = sp.symbols('a b c')
    num = 450*a**9*b**3*c**5 + 504*a**8*b**4*c**5 - 48*a**7*b**3*c**5 + 600*a**6*b**2*c**6 + 672*a**5*b**3*c**6 - 64*a**4*b**2*c**6
    den = -84*a*c - 75*b**2 - 84*b*c - 75*c**2 + 166
    g1 = a + b + c - 1
    g2 = a**2 + b**2 + c**2 - 4
    X = sp.Symbol('X')
    G = sp.groebner([g1, g2, num - den*X], a, b, c, X, order='lex')
    for poly in G.polynomials:
        if set(poly.free_symbols) <= {X}:
            if poly.degree() == 1:
                c0, c1 = poly.all_coeffs()
                P = -c1 / c0
                return sp.simplify(P)
            return poly
